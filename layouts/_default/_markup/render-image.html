{{- $src := .Destination -}} {{- $trimmed := strings.TrimPrefix "/" $src -}} {{- $diskPath := "" -}} {{- if and (hasPrefix $src "/") (not (hasPrefix $src "//")) -}} {{- if fileExists (printf "static/%s" $trimmed) -}} {{- $diskPath = printf "static/%s" $trimmed -}} {{- else if fileExists (printf "content/%s" $trimmed) -}} {{- $diskPath = printf "content/%s" $trimmed -}} {{- end -}} {{- end -}}
